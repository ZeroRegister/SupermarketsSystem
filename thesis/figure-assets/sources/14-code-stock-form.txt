async function saveMovement(){if(!movementForm.productId||!movementForm.reason.trim())return ElMessage.warning('Select a product and record a reason.');if(saving.value)return;saving.value=true;try{await api.post('/transactions',{...movementForm,idempotencyKey:movementRequestKey.value});ElMessage.success('Stock activity recorded with an audit trail.');movementDialog.value=false;loadAll()}catch(e:any){ElMessage.error(e?.response?.data?.message||'Could not record this stock activity.')}finally{saving.value=false}}
